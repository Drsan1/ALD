463[[nodiscard]] static inline bool vector_iterator_is_inside_span(const struct TSpan *aSpan, const struct TVectorIterator *aIter)
464 {
465 if(aIter && aIter->iVector)
466 {
467 assert(aIter->iVector->iValues);
468 const TVectorElement *elem_ptr = aIter->iVector->iValues + aIter->iPos;
469
470 return (elem_ptr >= aSpan->iVector.iValues) && (elem_ptr < aSpan->iVector.iValues + aSpan->iVector.iSize);
471 }
472 return false;
473 }
